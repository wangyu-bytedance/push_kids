"""Bounded, provider-neutral data for material extraction and alias-only decisions."""

from __future__ import annotations

from typing import Literal

from pydantic import BaseModel, ConfigDict, Field

from push_kids.agent_processing.contracts import DisplayKind, ReviewMethod

PIPELINE_REVISION = "staged-20261031-07"


class StageModel(BaseModel):
    model_config = ConfigDict(extra="forbid")


class ExtractedPoint(StageModel):
    subject: str = Field(min_length=1, max_length=40)
    name: str = Field(min_length=1, max_length=120)
    evidence: str = Field(min_length=1, max_length=200)
    materials: list[str] = Field(min_length=1, max_length=3)
    confidence: Literal["high", "medium", "low"]


class MaterialResult(StageModel):
    status: Literal["content", "non_learning", "unreadable"]
    needs_context: bool
    excerpt: str = Field(max_length=1200)
    items: list[ExtractedPoint] = Field(max_length=20)


class MergedPoint(StageModel):
    members: list[str] = Field(min_length=1, max_length=200)
    subject: str = Field(min_length=1, max_length=40)
    name: str = Field(min_length=1, max_length=120)
    category: str = Field(min_length=1, max_length=50)
    display_kind: DisplayKind
    review_method: ReviewMethod
    estimated_minutes: int = Field(ge=1, le=5)


class OmittedCandidates(StageModel):
    members: list[str] = Field(min_length=1, max_length=200)
    reason: Literal["no_learning_evidence", "missing_media_dependency", "indeterminate_object"]
    basis: str = Field(min_length=1, max_length=80)


class ConsolidatedResult(StageModel):
    summary: str = Field(min_length=1, max_length=60)
    items: list[MergedPoint] = Field(max_length=20)
    omissions: list[OmittedCandidates] = Field(max_length=20)
    uncertainties: list[str] = Field(max_length=20)


class MatchDecision(StageModel):
    candidate: str = Field(min_length=1, max_length=8)
    match: str | None = Field(max_length=8)
    uncertain: bool


class MatchingResult(StageModel):
    decisions: list[MatchDecision] = Field(min_length=1, max_length=20)


QualityIssueCode = Literal[
    "exercise_not_knowledge",
    "missing_media_dependency",
    "ordinary_scene",
    "vague_name",
    "should_merge",
    "over_merged",
    "unsupported_by_evidence",
    "invalid_review_scope",
]


class QualityIssue(StageModel):
    code: QualityIssueCode
    items: list[str] = Field(min_length=1, max_length=20)
    basis: str = Field(min_length=1, max_length=80)


class QualityReviewResult(StageModel):
    passed: bool
    issues: list[QualityIssue] = Field(max_length=20)


class ReuseDecision(StageModel):
    candidate: str = Field(min_length=1, max_length=8)
    match: str = Field(min_length=1, max_length=8)
    verdict: Literal["accept", "reject", "uncertain"]
    basis: str = Field(max_length=80)


class ReuseReviewResult(StageModel):
    decisions: list[ReuseDecision] = Field(min_length=1, max_length=20)
